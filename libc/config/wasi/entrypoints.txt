# The _Float16 builtin type is unavailable on WebAssembly targets without
# native half-precision support, so drop the float16 entrypoints (but keep
# the software-implemented bfloat16 ones) when the compiler lacks the type.
# Note: logbf16/ilogbf16/llogbf16 are logb-based float16 functions, not
# bfloat16 ones, so they are excluded as well.
if(NOT LIBC_TYPES_HAS_FLOAT16)
  list(FILTER TARGET_LIBM_ENTRYPOINTS EXCLUDE REGEX "[^b]f16|logbf16")
endif()

# setjmp/longjmp are only available when the WebAssembly exception-handling
# based sjlj lowering is enabled (see LIBC_WASM_ENABLE_SJLJ).
if(LIBC_WASM_ENABLE_SJLJ)
  list(APPEND TARGET_LIBC_ENTRYPOINTS
    libc.src.setjmp.setjmp
    libc.src.setjmp.longjmp
    libc.src.setjmp.sigsetjmp
    libc.src.setjmp.siglongjmp
  )
endif()

set(TARGET_LLVMLIBC_ENTRYPOINTS
  ${TARGET_LIBC_ENTRYPOINTS}
  ${TARGET_LIBM_ENTRYPOINTS}
)
